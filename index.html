  let state = { uid: null, budget: 100, points: 0, squad: [], captainId: null, leagueId: "", quinielaPicks: {}, quinielaHistory: [] };

  function generateBalancedSquad() {

    const squad = [];

    const pool = [...POOL];

    const pick = (pos, tier) => {

      const idx = pool.findIndex(p => p.pos === pos && p.tier === tier);

      if (idx !== -1) return pool.splice(idx, 1)[0];

      const fallback = pool.findIndex(p => p.pos === pos);

      return fallback !== -1 ? pool.splice(fallback, 1)[0] : null;

    };

    squad.push(pick('POR', 'good'), pick('DEF', 'star'), pick('DEF', 'good'), pick('DEF', 'decent'), pick('DEF', 'weak'));

    squad.push(pick('MED', 'star'), pick('MED', 'good'), pick('MED', 'decent'), pick('MED', 'weak'));

    squad.push(pick('DEL', 'good'), pick('DEL', 'decent'));

    squad.push(pick('POR', 'weak') || pick('POR', 'decent'), pick('DEF', 'weak') || pick('DEF', 'decent'));

    squad.push(pick('MED', 'weak') || pick('MED', 'decent'), pick('DEL', 'weak') || pick('DEL', 'decent'));

    squad.push(pick('ENT', 'star') || pick('ENT', 'good') || pick('ENT', 'decent'));

    return squad.filter(Boolean);

  }


  async function loadUserData(uid) {

    const userRef = doc(db, "users", uid);

    const userSnap = await getDoc(userRef);

    if (userSnap.exists()) {

      state = { ...state, ...userSnap.data() };

    } else {

      state = { ...state, uid: uid, squad: generateBalancedSquad(), captainId: state.squad.find(p => p.tier === 'star' && p.pos !== 'ENT')?.id || null };

      await setDoc(userRef, state);

    }

    updateBudgetUI();

    buildPitch();

    buildSquad();

    buildMarket();

    buildSell();

    buildMatches();

  }


  async function saveState() {

    if (currentUser) {

      await updateDoc(doc(db, "users", currentUser.uid), state);

      updateBudgetUI();

    }

  }

  function toastMsg(msg) {

    const t = document.getElementById('toast');

    t.textContent = msg;

    t.classList.add('show');

    setTimeout(() => t.classList.remove('show'), 2400);

  }


  function tierLabel(t) {

    if (t === 'star') return '<span class="star-badge">★</span>';

    if (t === 'good') return '<span class="good-badge">Bueno</span>';

    return '';

  }


  async function setCaptain(id) {

    const p = state.squad.find(x => x.id === id);

    if (!p || p.pos === 'ENT') { toastMsg('Solo jugadores de campo pueden ser capitán'); return; }

    state.captainId = id;

    toastMsg(`🎯 ${p.name} es ahora el Capitán (puntos x2)`);

    await saveState();

    buildPitch();

    buildSquad();

  }


  function buildPitch() {

    const zones = { POR: [], DEF: [], MED: [], DEL: [], BEN: [] };

    state.squad.forEach((p, i) => {

      if (i >= 16) zones.BEN.push({ p, i });

      else zones[p.pos].push({ p, i });

    });

    ['POR', 'DEF', 'MED', 'DEL', 'BEN'].forEach(z => {

      const el = document.getElementById('p-' + z);

      if (!el) return;

      el.innerHTML = '';

      zones[z].forEach(({ p, i }) => {

        const bench = z === 'BEN';

        const isCaptain = state.captainId === p.id && !bench;

        el.innerHTML += `<div class="pp ${bench ? 'bench' : ''} ${isCaptain ? 'captain' : ''}" onclick="${bench ? '' : `window.setCaptain(${p.id})`}">

          <div class="pp-flag">${flag(p.nat)}</div>

          <div class="pp-name">${p.name.split(' ')[0]}</div>

          <div class="pp-pts">${p.pts}</div>

          <div class="pp-pos">${p.pos}${bench ? ' (banco)' : ''} ${isCaptain ? '🎯' : ''}</div>

        </div>`;

      });

    });

  }

  async function buyPlayer(id, price, name) {

    if (state.budget < price) { toastMsg('Presupuesto insuficiente'); return; }

    if (state.squad.length >= 17) { toastMsg('Plantilla completa (17 máx)'); return; }

    state.budget -= price;

    state.squad.push(POOL.find(p => p.id === id));

    toastMsg(`¡${name} fichado!`);

    await saveState();

    buildPitch(); buildSquad(); buildMarket(); buildSell();

  }


  function buildSell() {

    const el = document.getElementById('sell-list');

    el.innerHTML = '';

    state.squad.forEach((p, i) => {

      const sellPrice = Math.floor(p.price * 0.9);

      el.innerHTML += `<div class="market-row">

        <span class="pos pos-${p.pos}">${p.pos}</span>

        <div class="player-info"><div class="player-name">${flag(p.nat)} ${p.name}</div></div>

        <button class="btn-action btn-sell" onclick="window.sellPlayer(${i}, ${sellPrice}, '${p.name}')">+${sellPrice}M</button>

      </div>`;

    });

  }


  async function sellPlayer(index, price, name) {

    state.budget += price;

    if (state.captainId === state.squad[index].id) state.captainId = null;

    state.squad.splice(index, 1);

    toastMsg(`${name} vendido`);

    await saveState();

    buildPitch(); buildSquad(); buildMarket(); buildSell();

  }

  function pickQ(i, v) {

    ['h','d','a'].forEach(x => document.getElementById(`${x}-${i}`)?.classList.remove('sel'));

    document.getElementById(`${v}-${i}`)?.classList.add('sel');

    state.quinielaPicks[i] = v;

  }


  async function submitQ() {

    if (Object.keys(state.quinielaPicks).length < 4) { toastMsg('Completa las 4 predicciones'); return; }

    let pts = Math.floor(Math.random() * 50) + 30; 

    state.points += pts;

    state.quinielaHistory.unshift(`Jornada 1: +${pts} pts`);

    document.getElementById('btn-submit').disabled = true;

    toastMsg(`¡Quiniela enviada! +${pts} pts`);

    await saveState();

  }
